  <script>
    // ===== 유틸 =====
    function toYMD(date){
      if(!date) return '';
      const d = new Date(date);
      const y = d.getFullYear();
      const m = String(d.getMonth()+1).padStart(2,'0');
      const dd = String(d.getDate()).padStart(2,'0');
      return `${y}.${m}.${dd}`;
    }
    function weekdayK(date){
      const d = new Date(date);
      return ['일','월','화','수','목','금','토'][d.getDay()];
    }
    function fmt(date){ if(!date) return ''; return `${toYMD(date)}(${weekdayK(date)})`; }
    function padLeft(text, total){
      const len = text.length; const n = Math.max(0, total - len);
      return ' '.repeat(n) + text;
    }

    // ===== 요소 =====
    const $ = (id) => document.getElementById(id);
    const fields = ['작성일자','회사명','지회코드','상조회원여부','성명','직함','관계','고인성함','별세일자','발인일자','장례식장명','주소','사무실전화','휴대전화표기','마음전하실곳','기관명'];

    function build(){
      const data = Object.fromEntries(fields.map(k=>[k, k==='상조회원여부'? $(k).checked : $(k).value]));

      const title = padLeft('訃     告', 94);
      const dateLine = padLeft(fmt(data.작성일자), 109);
      const prefix = data.상조회원여부 ? '우리시회 회원이시며 상조회원이신' : '우리시회 회원이신';
      const first = `${prefix} ${data.회사명}(${data.지회코드}) ${data.성명} ${data.직함}님의 ${data.관계} ${data.고인성함}님께서 별세하셨기에 다음과 같이 알려드립니다.`;

      const lines = [
        `■ 별세일자 : ${fmt(data.별세일자)}`,
        `■ 발인일자 : ${fmt(data.발인일자)}`,
        `■ 빈    소 : ${data.장례식장명}`,
        `               (${data.주소})`,
        `■ 연락처 : ${data.사무실전화} (사무실)`,
        `              ${data.휴대전화표기}`,
        `※ 마음 전하실 곳 : [${data.마음전하실곳}]`,
        `※ 일괄 전송으로 인하여 관련 없으신 회원께서는 양해하여 주시기 바랍니다.`,
        padLeft(data.기관명, 99)
      ];

      $('output').textContent = [title, dateLine, first, ...lines].join('
');
    }

    // 초기값: 오늘 날짜 자동 채움
    const today = new Date();
    $('작성일자').valueAsDate = today;
    $('별세일자').valueAsDate = today;
    $('발인일자').valueAsDate = today;

    // 이벤트 바인딩
    fields.forEach(k=>{ const el=$(k); if(!el) return; el.addEventListener('input', build); el.addEventListener('change', build); });

    $('copyBtn').onclick = async () => {
      try {
        await navigator.clipboard.writeText($('output').textContent);
        alert('복사되었습니다. 문자/카톡/문서에 붙여넣기 하세요.');
      } catch(e){ alert('복사 실패: HTTPS 환경 또는 브라우저 권한을 확인하세요.'); }
    };
    $('printBtn').onclick = () => window.print();
    $('exampleBtn').onclick = () => {
      $('회사명').value = '(주)성진전기';
      $('지회코드').value = '대구-00927';
      $('성명').value = '박일근';
      $('직함').value = '대표이사';
      $('관계').value = '장모';
      $('고인성함').value = '김춘자';
      $('장례식장명').value = '대구성심병원장례식장 특실';
      $('주소').value = '대구광역시 남구 대명로 14';
      $('사무실전화').value = '053-783-5598';
      $('휴대전화표기').value = '010-3803-7162 (박일근 대표)';
      $('마음전하실곳').value = '농협은행 / 8411-2656-0113-93 박일근';
      build();
    };

    build();
  </script>
